# 1. GET запит (картинки немає в кеші, завантажується з http.cat)
curl -i http://127.0.0.1:3000/200

# 2. Повторний GET запит (віддається з кешу, заголовок X-Cache-Source: cache)
curl -i http://127.0.0.1:3000/200

# 3. PUT запит (збереження власної картинки, наприклад, з файлу image.jpg)
curl -i -X PUT --data-binary "@image.jpg" http://127.0.0.1:3000/418

# 4. GET після PUT (перевірка, що віддається наша збережена картинка)
curl -i http://127.0.0.1:3000/418

# 5. DELETE запит (успішне видалення з кешу)
curl -i -X DELETE http://127.0.0.1:3000/418

# 6. DELETE неіснуючого файлу (помилка 404)
curl -i -X DELETE http://127.0.0.1:3000/999

# 7. GET неіснуючого коду стану (помилка 404)
curl -i http://127.0.0.1:3000/999

# 8. Запит із недозволеним методом POST (очікуємо 405 Method Not Allowed із заголовком Allow)
curl -i -X POST http://127.0.0.1:3000/200
